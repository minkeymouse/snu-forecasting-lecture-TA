# 국가별 분기 대비 성장률을 연율화하여 계산
df_growth = (
    df.set_index(['date','country'])[vars_to_use]
      .sort_index()
      .groupby('country')
      .pct_change()   # 전분기 대비 % 변화
      .mul(400)       # 연율화
      .dropna()
      .reset_index()
)

# 와이드 패널로 피벗(MultiIndex 컬럼)
panel = df_growth.pivot(
    index='date',
    columns='country',
    values=vars_to_use
)

# 결측은 제거하지 않고 전/후 값으로 채움
panel = panel.interpolate().ffill().bfill()

# 상수가 된 시계열은 제거
panel = panel.loc[:, panel.std() > 0]

# 이제 MultiIndex를 단일 문자열로 평탄화: "Y_CAN", "C_DEU" 등
panel.columns = [f"{var}_{ctry}" for var, ctry in panel.columns]

# 표준화(평균=0, 표준편차=1)
endog = (panel - panel.mean()) / panel.std()

# 2. DynamicFactorMQ 설정 및 적합
# ---------------------------------
k_factors  = 3   # 공통 요인 개수
factor_ord = 4   # 요인 동학의 VAR 차수

model = sm.tsa.DynamicFactorMQ(
    endog,
    factors=k_factors,
    factor_orders=factor_ord,
    idiosyncratic_ar1=True,
    standardize=False
)
